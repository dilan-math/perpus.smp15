Kode Apps Script (Mendukung Edit Manual di Sheet)

Agar edit manual di Spreadsheet tidak ter-reset:

Ganti kode di Extensions → Apps Script dengan kode baru di bawah ini, lalu klik Deploy → Manage deployments → Edit (ikon pensil) → Version: New version → Deploy. Data akan otomatis tersusun rapi menjadi 3 tab tabel: Transaksi, Dompet, dan Anggaran.

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  if (e && e.parameter && e.parameter.action === 'save' && e.parameter.payload) {
    saveAllTables(ss, JSON.parse(e.parameter.payload));
    return buildOut(e, { status: 'ok' });
  }
  return buildOut(e, readAllTables(ss));
}
function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var body = e.postData ? e.postData.contents : '';
  if (body) saveAllTables(ss, JSON.parse(body));
  return ContentService.createTextOutput(JSON.stringify({ status: 'ok' }))
    .setMimeType(ContentService.MimeType.JSON);
}
function readAllTables(ss) {
  var sTx = ss.getSheetByName('Riwayat_Transaksi') || ss.getSheetByName('Transaksi') || ss.getSheets()[0];
  var sWal = ss.getSheetByName('Dompet');
  var sBud = ss.getSheetByName('Anggaran');
  var sMeta = ss.getSheetByName('Sakusaku_DB') || ss.getSheetByName('SakusakuDB');
  if (!sTx && !sWal && sMeta) {
    var legacy = sMeta.getRange('A1').getValue();
    return legacy ? JSON.parse(legacy) : { status: 'empty' };
  }
  if (!sTx && !sWal) return { status: 'empty' };

  var wallets = [];
  if (sWal && sWal.getLastRow() > 1) {
    sWal.getRange(2, 1, sWal.getLastRow() - 1, 5).getValues().forEach(function(r) {
      if (r[1]) wallets.push({
        id: String(r[0] || ('w_' + Math.random().toString(36).slice(2, 8))),
        name: String(r[1]), type: String(r[2] || 'cash'),
        initialBalance: Number(r[3]) || 0, color: String(r[4] || 'indigo')
      });
    });
  }

  var transactions = [];
  if (sTx && sTx.getLastRow() > 1) {
    var numCols = Math.max(7, sTx.getLastColumn());
    sTx.getRange(2, 1, sTx.getLastRow() - 1, numCols).getValues().forEach(function(r, idx) {
      // Deteksi otomatis apakah kolom A adalah Tanggal (7 kolom) atau ID (8 kolom)
      var hasIdCol = (String(r[0] || '').indexOf('tx_') === 0);
      var rawDate = hasIdCol ? r[1] : r[0];
      var rawType = hasIdCol ? r[2] : r[1];
      var rawCat  = hasIdCol ? r[3] : r[2];
      var rawWal  = hasIdCol ? r[4] : r[3];
      var rawToW  = hasIdCol ? r[5] : r[4];
      var rawAmt  = hasIdCol ? r[6] : r[5];
      var rawNote = hasIdCol ? r[7] : r[6];

      if (rawDate || rawAmt) {
        var dt = '';
        if (rawDate instanceof Date) {
          dt = Utilities.formatDate(rawDate, ss.getSpreadsheetTimeZone(), 'yyyy-MM-dd');
        } else {
          var sDt = String(rawDate || '').trim();
          var mSlash = sDt.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/);
          if (mSlash) {
            dt = mSlash[3] + '-' + ('0' + mSlash[2]).slice(-2) + '-' + ('0' + mSlash[1]).slice(-2);
          } else {
            dt = sDt.slice(0, 10);
          }
        }
        var cleanAmt = Number(String(rawAmt || 0).replace(/[^\d-]/g, '')) || 0;
        var cleanToW = String(rawToW || '').trim();
        if (cleanToW === '-') cleanToW = '';

        transactions.push({
          id: hasIdCol ? String(r[0]) : ('tx_row_' + (idx + 2)),
          date: dt,
          type: String(rawType || 'Pengeluaran').trim(),
          category: String(rawCat || 'Lainnya').trim(),
          walletId: String(rawWal || 'Dompet Tunai').trim(),
          toWalletId: cleanToW || null,
          amount: cleanAmt,
          note: String(rawNote || '').trim()
        });
      }
    });
  }

  var budgets = [];
  if (sBud && sBud.getLastRow() > 1) {
    sBud.getRange(2, 1, sBud.getLastRow() - 1, 2).getValues().forEach(function(r) {
      if (r[0]) budgets.push({ categoryId: String(r[0]), limit: Number(String(r[1] || 0).replace(/[^\d-]/g, '')) || 0 });
    });
  }

  var customCategories = { expense: [], income: [] };
  if (sMeta && sMeta.getRange('A2').getValue()) {
    try { customCategories = JSON.parse(sMeta.getRange('A2').getValue()); } catch (err) {}
  }

  return { status: 'ok', wallets: wallets, transactions: transactions, budgets: budgets, customCategories: customCategories };
}
function saveAllTables(ss, data) {
  var sWal = ss.getSheetByName('Dompet') || ss.insertSheet('Dompet');
  sWal.clearContents();
  var wRows = [['ID', 'Nama Dompet', 'Jenis', 'Saldo Awal', 'Warna']];
  (data.wallets || []).forEach(function(w) { wRows.push([w.id, w.name, w.type, Number(w.initialBalance) || 0, w.color]); });
  sWal.getRange(1, 1, wRows.length, 5).setValues(wRows);

  // Gunakan tab 'Riwayat_Transaksi' sebagai sumber utama
  var sTx = ss.getSheetByName('Riwayat_Transaksi') || ss.getSheetByName('Transaksi') || ss.getSheets()[0];
  sTx.clearContents();
  var walletMap = {};
  (data.wallets || []).forEach(function(w) { walletMap[w.id] = w.name; });
  var catMap = {
    makan: 'Makan & Minum', belanja: 'Belanja', transport: 'Transportasi',
    tagihan: 'Tagihan & Utilitas', hiburan: 'Hiburan', kesehatan: 'Kesehatan',
    pendidikan: 'Pendidikan', lainnya_exp: 'Lainnya', gaji: 'Gaji Bulanan',
    bonus: 'Bonus & THR', investasi: 'Investasi', lainnya_inc: 'Pemasukan Lain'
  };
  ((data.customCategories && data.customCategories.expense) || []).forEach(function(c) { catMap[c.id] = c.name; });
  ((data.customCategories && data.customCategories.income) || []).forEach(function(c) { catMap[c.id] = c.name; });

  var tRows = [['Tanggal', 'Jenis', 'Kategori', 'Dompet Asal', 'Dompet Tujuan', 'Nominal (Rp)', 'Catatan']];
  (data.transactions || []).forEach(function(t) {
    var jenisIndo = t.type === 'income' ? 'Pemasukan' : (t.type === 'transfer' ? 'Transfer' : 'Pengeluaran');
    var katIndo = catMap[t.category] || t.category || 'Lainnya';
    var wAsal = walletMap[t.walletId] || t.walletId || 'Dompet Tunai';
    var wTujuan = t.toWalletId ? (walletMap[t.toWalletId] || t.toWalletId) : '-';
    tRows.push([t.date, jenisIndo, katIndo, wAsal, wTujuan, Number(t.amount) || 0, t.note || '']);
  });
  sTx.getRange(1, 1, tRows.length, 7).setValues(tRows);

  var sBud = ss.getSheetByName('Anggaran') || ss.insertSheet('Anggaran');
  sBud.clearContents();
  var bRows = [['ID Kategori', 'Batas Anggaran']];
  (data.budgets || []).forEach(function(b) { bRows.push([b.categoryId, Number(b.limit) || 0]); });
  sBud.getRange(1, 1, bRows.length, 2).setValues(bRows);

  if (data.customCategories) {
    var sMeta = ss.getSheetByName('Sakusaku_DB') || ss.getSheetByName('SakusakuDB') || ss.insertSheet('Sakusaku_DB');
    sMeta.getRange('A2').setValue(JSON.stringify(data.customCategories));
  }
}
function buildOut(e, obj) {
  var json = JSON.stringify(obj);
  if (e && e.parameter && e.parameter.callback) {
    return ContentService.createTextOutput(e.parameter.callback + '(' + json + ')').setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(json).setMimeType(ContentService.MimeType.JSON);
}
